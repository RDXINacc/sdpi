%% ============================================================
% PRACTICAL 10: STACKING & STOLT (F-K) MIGRATION
%
% 1. Stack seismic data after required preprocessing (NMO/mutes).
% 2. Apply Stolt migration in f-k domain using constant velocity.
% 3. Apply gain to enhance visibility and compare results.
% ============================================================
clear;
clc;
close all;

%% 1. LOAD DATA & STACKING
if exist('oz_04_Practical7_6_results.mat', 'file') == 2
    loaded = load('oz_04_Practical7_6_results.mat');
    nmo_data = loaded.nmo_data; % Preprocessed NMO gather
    t        = loaded.t;
    dt       = loaded.dt;
    
    if size(nmo_data, 2) <= 48
        % Department stack equivalent (Instruction 1)
        single_stack = sum(nmo_data, 2) ./ max(sum(nmo_data ~= 0, 2), 1);
        
        nx = 96;
        dx = 25; % CMP spacing (m)
        x  = (0:nx-1) * dx;
        stacked_section = repmat(single_stack, 1, nx);
        
        % Synthetic dipping reflector to evaluate dip migration
        v_dip = 2200;
        for ix = 1:nx
            t_dip = 1.8 + (x(ix) - mean(x)) * sind(20) / (v_dip / 2);
            idx = round(t_dip / dt);
            if idx > 10 && idx < length(t)-10
                stacked_section(idx-2:idx+2, ix) = stacked_section(idx-2:idx+2, ix) + ...
                    [0.2; 0.6; 1.0; 0.6; 0.2] * max(single_stack);
            end
        end
    else
        stacked_section = nmo_data;
        dt = loaded.dt;
        t  = loaded.t;
        dx = 25;
        nx = size(stacked_section, 2);
        x  = (0:nx-1) * dx;
    end
else
    % Fallback synthetic stacked section
    dt = 0.004;
    t  = (0:1500)' * dt;
    nt = length(t);
    dx = 25;
    nx = 100;
    x  = (0:nx-1) * dx;
    stacked_section = zeros(nt, nx);
    
    for ix = 1:nx
        i1 = round(1.0 / dt);
        stacked_section(i1, ix) = 1.0;
        t_dip = 1.5 + (ix - nx/2) * dx * sind(15) / (2000 / 2);
        i2 = round(t_dip / dt);
        if i2 > 0 && i2 <= nt
            stacked_section(i2, ix) = 1.0;
        end
    end
end

[nt, nx] = size(stacked_section);
fprintf('Stacked section: %d time samples x %d traces\n', nt, nx);

%% 2. MIGRATION PARAMETERS (CONSTANT VELOCITY)
% Exploding reflector model uses half-velocity: v_mig = v / 2
v_medium = 2200;         % Medium velocity (m/s)
v_mig    = v_medium / 2; % Migration velocity (m/s)
fprintf('Migration velocity (half-velocity) = %.1f m/s\n', v_mig);

%% 3. SPATIAL BOUNDARY TAPERING & 2D FFT
% Taper trace boundaries along x to suppress Fourier edge-diffraction smiles
taper_x = tukeywin(nx, 0.15)'; % 15% cosine taper
stacked_tapered = stacked_section .* repmat(taper_x, nt, 1);

nfft_t = 2^nextpow2(nt * 2);
nfft_x = 2^nextpow2(nx * 2);

% Frequency axes
fs = 1 / dt;
f  = (-nfft_t/2 : (nfft_t/2 - 1)) * (fs / nfft_t);
wz = 2 * pi * f(:); % Migrated vertical frequency axis (nfft_t x 1 column)

ks = 1 / dx;
kx_axis = (-nfft_x/2 : (nfft_x/2 - 1)) * (ks / nfft_x);
kx = 2 * pi * kx_axis(:)'; % Spatial wavenumber axis (1 x nfft_x row)

% Forward 2D FFT: (t, x) -> (w, kx)
D_wk = fftshift(fft2(stacked_tapered, nfft_t, nfft_x));

%% 4. STOLT INVERSE DISPERSION MAPPING
% Solves for unmigrated temporal frequency w_input from vertical frequency wz:
% w_input = sign(wz) * sqrt(wz^2 + (v_mig * kx)^2) evaluated where valid,
% moving events updip and steepening dips.
M_wk = zeros(nfft_t, nfft_x);

fprintf('Applying Stolt migration mapping...\n');
for ik = 1:nfft_x
    k_val = kx(ik);
    
    % Propagating condition: non-evanescent wavefield
    arg = wz.^2 - (v_mig * k_val)^2;
    valid = arg > 0;
    
    w_input = zeros(size(wz));
    w_input(valid) = sign(wz(valid)) .* sqrt(arg(valid));
    
    % Obliquity / Jacobian scale factor: d(w)/d(wz)
    jacobian = zeros(size(wz));
    jacobian(valid) = abs(wz(valid)) ./ max(abs(w_input(valid)), eps);
    
    % 1D interpolation along unmigrated frequency column
    interpolated = interp1(wz, D_wk(:, ik), w_input, 'linear', 0);
    M_wk(:, ik) = interpolated .* jacobian;
end

%% 5. INVERSE 2D FFT: (w, kx) -> (t, x)
migrated_padded = real(ifft2(ifftshift(M_wk)));
migrated_data   = migrated_padded(1:nt, 1:nx);

%% 6. GAIN ENHANCEMENT (AGC - AUTOMATIC GAIN CONTROL)
agc_window_len = round(0.3 / dt); % 300 ms window
h_win = hamming(agc_window_len);
h_win = h_win / sum(h_win);

stacked_gained  = zeros(nt, nx);
migrated_gained = zeros(nt, nx);

for ix = 1:nx
    env1 = sqrt(conv(stacked_section(:, ix).^2, h_win, 'same') + 1e-12);
    stacked_gained(:, ix) = stacked_section(:, ix) ./ env1;
    
    env2 = sqrt(conv(migrated_data(:, ix).^2, h_win, 'same') + 1e-12);
    migrated_gained(:, ix) = migrated_data(:, ix) ./ env2;
end

%% 7. PLOT AND COMPARE RESULTS
figure('Name', 'Practical 10: Stolt Migration', 'NumberTitle', 'off', 'Position', [80 80 1200 600]);

% Before Migration (Stacked Section)
subplot(1, 2, 1);
imagesc(x, t, stacked_gained);
set(gca, 'YDir', 'reverse');
xlabel('Distance / CMP Position (m)');
ylabel('Two-Way Traveltime (s)');
title('Before Migration (Stacked Section with Gain)');
colormap(gray);
colorbar;
caxis([-3 3]);
axis tight;

% After Stolt Migration
subplot(1, 2, 2);
imagesc(x, t, migrated_gained);
set(gca, 'YDir', 'reverse');
xlabel('Distance / CMP Position (m)');
ylabel('Two-Way Traveltime (s)');
title('After Stolt Migration (Migrated Section with Gain)');
colormap(gray);
colorbar;
caxis([-3 3]);
axis tight;

sgtitle(sprintf('Practical 10: Comparison Before and After Stolt F-K Migration (v = %d m/s)', v_medium));

%% 8. SAVE OUTPUT
save('practical10_stolt_results.mat', 't', 'x', 'stacked_section', 'migrated_data', 'v_medium');
fprintf('Processing completed and saved to practical10_stolt_results.mat\n');
