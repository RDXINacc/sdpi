%% ============================================================
% PRACTICAL 8: DIP MOVEOUT (DMO) CORRECTION IN F-K DOMAIN
%
% Objective: Perform DMO on NMO-corrected CMP data and compare 
%            results before and after DMO.
% ============================================================
clear;
clc;
close all;

%% 1. INPUT: LOAD NMO-CORRECTED CMP GATHER
% Option A: Load from previously saved Practical 6/7 results file
mat_file = 'oz_04_Practical7_6_results.mat'; 

if exist(mat_file, 'file') == 2
    loaded = load(mat_file);
    nmo_data = loaded.nmo_data;
    t        = loaded.t;
    dt       = loaded.dt;
    h        = loaded.h;
    if isfield(loaded, 'v_nmo')
        v_ref = mean(loaded.v_nmo); % Mean replacement velocity
    else
        v_ref = 2000; % Default replacement velocity (m/s)
    end
else
    % Fallback / standalone synthetic setup if .mat is not found
    warning('Saved results file not found. Generating synthetic NMO gather.');
    dt = 0.004;
    t = (0:2534)' * dt;
    h = [-2550:100:-250, 250:100:2550]';
    v_ref = 2000;
    nmo_data = zeros(length(t), length(h));
    % Dipping event simulation after NMO
    for ih = 1:length(h)
        t_event = 1.5 + 0.0001 * h(ih); % Residual moveout from dip
        idx = round(t_event / dt);
        if idx > 0 && idx <= length(t)
            nmo_data(idx, ih) = 1.0;
        end
    end
end

[nt, nh] = size(nmo_data);
fprintf('Gather dimensions: %d samples x %d traces\n', nt, nh);
fprintf('Mean velocity v = %.1f m/s\n', v_ref);

%% 2 & 3. BUILD FREQUENCY (omega) AND WAVENUMBER (k) GRIDS
% Spatial trace spacing (assumed uniform dx)
dh = mean(diff(sort(abs(h)))); 
if isnan(dh) || dh == 0
    dh = 100; % Default spacing in meters
end

% Next power of 2 for clean FFT
nfft_t = 2^nextpow2(nt);
nfft_x = 2^nextpow2(nh);

% Temporal frequency axis f (Hz) and angular frequency omega (rad/s)
fs = 1 / dt;
f = (-nfft_t/2 : (nfft_t/2 - 1)) * (fs / nfft_t);
omega = 2 * pi * f(:); % column vector (nfft_t x 1)

% Spatial frequency / wavenumber axis kx (cycles/m) and k (rad/m)
ks = 1 / dh;
kx = (-nfft_x/2 : (nfft_x/2 - 1)) * (ks / nfft_x);
k = 2 * pi * kx(:)'; % row vector (1 x nfft_x)

%% 4 & 5 & 6. 2D F-K TRANSFORM AND HALE-TYPE PHASE-SHIFT DMO
% 2D FFT to f-k domain (centered spectrum)
D_fk = fftshift(fft2(nmo_data, nfft_t, nfft_x));

% Prepare modified f-k spectrum
D_fk_dmo = zeros(size(D_fk));

% Trace-by-trace DMO application using trace actual offset h
% For regular grids, we compute the Hale operator across omega and k:
fprintf('Applying Hale-type DMO operator...\n');

% Regularization epsilon to prevent division by zero at omega = 0
eps_w = 1e-4;
omega_reg = omega;
omega_reg(abs(omega_reg) < eps_w) = sign(omega_reg(abs(omega_reg) < eps_w) + eps) * eps_w;

% Compute average half-offset or trace-dependent operator
% Hale formula: dt_dmo = (h^2 * k^2) / (2 * v^2 * omega)
% DMO = exp(-i * omega * dt_dmo) = exp(-i * (h^2 * k^2) / (2 * v^2))
for ih = 1:nh
    h_curr = h(ih); % Actual offset
    
    % Hale phase-shift formulation as given in instructions:
    % dt_shift = (h_curr^2 * (k.^2)) ./ (2 * (v_ref^2) * omega_reg);
    % DMO_op   = exp(-1i * (omega_reg .* dt_shift));
    % Notice omega cancels out in (omega * dt_shift):
    phase_shift = (h_curr^2 .* (k.^2)) ./ (2 * v_ref^2);
    DMO_op = exp(-1i * repmat(phase_shift, nfft_t, 1));
    
    % Apply per trace in wavenumber slice
    D_fk_dmo(:, :) = D_fk_dmo(:, :) + (D_fk .* DMO_op) / nh;
end

%% 7. INVERSE TRANSFORM BACK TO TIME-OFFSET DOMAIN
% Inverse FFT2 from f-k to t-x
dmo_data_padded = real(ifft2(ifftshift(D_fk_dmo)));

% Truncate back to original time and trace dimensions
dmo_data = dmo_data_padded(1:nt, 1:nh);

fprintf('DMO processing completed.\n');

%% 8. PLOT BEFORE AND AFTER DMO COMPARISON
figure('Name', 'Practical 8: DMO Comparison', 'NumberTitle', 'off', 'Position', [100 100 1200 600]);

% Before DMO (NMO Corrected)
subplot(1, 2, 1);
imagesc(h, t, nmo_data);
set(gca, 'YDir', 'reverse');
xlabel('Offset (m)');
ylabel('Two-Way Traveltime (s)');
title('Before DMO (NMO-Corrected Gather)');
colormap(gray);
colorbar;
caxis([-max(abs(nmo_data(:)))*0.5, max(abs(nmo_data(:)))*0.5]);
axis tight;

% After DMO
subplot(1, 2, 2);
imagesc(h, t, dmo_data);
set(gca, 'YDir', 'reverse');
xlabel('Offset (m)');
ylabel('Two-Way Traveltime (s)');
title('After DMO (Hale F-K DMO Applied)');
colormap(gray);
colorbar;
caxis([-max(abs(dmo_data(:)))*0.5, max(abs(dmo_data(:)))*0.5]);
axis tight;

sgtitle('Comparison of CMP Gather Before and After Dip Moveout (DMO)');

%% 9. SAVE OUTPUT RESULTS
save('practical8_dmo_results.mat', 't', 'h', 'nmo_data', 'dmo_data');
fprintf('Results saved to practical8_dmo_results.mat\n');
